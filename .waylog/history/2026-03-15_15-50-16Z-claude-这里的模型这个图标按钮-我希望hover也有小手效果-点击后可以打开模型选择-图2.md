export function assignModelToGroup(
  modelName: string,
  vendorName?: string,
): string {
  const lowerModelName = modelName.toLowerCase();
  
  if (!vendorName || vendorName === "自定义模型" || vendorName === "Custom") {
    return "自定义模型";
  }
  
  const vendorGroup = MODEL_GROUPS.find((group) => group.name === vendorName);
  if (vendorGroup) {
    return vendorGroup.name;
  }
  
  for (const group of MODEL_GROUPS) {
    if (group.name === "自定义模型") continue;
    for (const keyword of group.keywords) {
      if (lowerModelName.includes(keyword.toLowerCase())) {
        return group.name;
      }
    }
  }
  
  return "OpenAI";
}
