```typescript
async function buildAiHubMixVideoRequestBody(model, options) {
  const payload = buildAiHubMixVideoPayload(model, options);
  const imageUrl = String((payload as any).image_url || "");

  if (!imageUrl) {
    return payload;  // 纯文本视频生成，返回 JSON
  }

  // 图生视频：构建 FormData 上传参考图
  const formData = new FormData();
  formData.append("model", model);
  formData.append("prompt", prompt);
  
  const imageBlob = await fetch(imageUrl).then((res) => res.blob());
  formData.append("input_reference", imageBlob, `reference.${extension}`);
  
  return formData;
}
```

async function pollAiHubMixTask(pollingUrl, headers, controller, onProgress) {
